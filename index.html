function takeTeamsSnapshot() { const container = document.getElementById('teams-capture-container'); const btn = event.currentTarget; const originalHTML = btn.innerHTML; btn.innerHTML = ` Génération...`; btn.disabled = true; // 1. Masquer temporairement les cotes le temps du rendu html2canvas const wasHidden = document.body.classList.contains('hide-ratings'); if (!wasHidden) { document.body.classList.add('hide-ratings'); } html2canvas(container, { scale: 2, useCORS: true, backgroundColor: '#f1f5f9' }).then(canvas => { canvas.toBlob(blob => { const item = new ClipboardItem({ "image/png": blob }); navigator.clipboard.write([item]).then(() => { alert("📸 Photo copiée dans le presse-papier ! Prête à coller sur Facebook."); }).catch(() => { const link = document.createElement('a'); link.download = `alignement-${new Date().toISOString().split('T')[0]}.png`; link.href = canvas.toDataURL(); link.click(); alert("📸 Image téléchargée sur votre appareil !"); }); // 2. Rétablir l'affichage normal des cotes à l'écran if (!wasHidden) { document.body.classList.remove('hide-ratings'); } btn.innerHTML = originalHTML; btn.disabled = false; }); }).catch(err => { console.error(err); if (!wasHidden) { document.body.classList.remove('hide-ratings'); } btn.innerHTML = originalHTML; btn.disabled = false; }); }
